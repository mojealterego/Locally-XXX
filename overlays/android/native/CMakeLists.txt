cmake_minimum_required(VERSION 3.22)
project(omega_llama_jni LANGUAGES C CXX)

set(OMEGA_LLAMA_PIN "418fe97b5f01b11eeb89f30d8a0459433edbf1d2")
if(NOT ANDROID)
    message(FATAL_ERROR "Production CMake target requires Android NDK. For host tests run tests/native/test_llama_jni_host.sh")
endif()
if(NOT DEFINED OMEGA_LLAMA_SOURCE OR NOT EXISTS "${OMEGA_LLAMA_SOURCE}/CMakeLists.txt")
    message(FATAL_ERROR "Supply a complete local llama.cpp checkout at OMEGA_LLAMA_SOURCE")
endif()
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${OMEGA_LLAMA_SOURCE}"
    OUTPUT_VARIABLE LLAMA_ACTUAL OUTPUT_STRIP_TRAILING_WHITESPACE
    RESULT_VARIABLE LLAMA_GIT_STATUS ERROR_QUIET)
if(NOT LLAMA_GIT_STATUS EQUAL 0 OR NOT LLAMA_ACTUAL STREQUAL OMEGA_LLAMA_PIN)
    message(FATAL_ERROR "llama.cpp checkout is not the pinned audited commit")
endif()
execute_process(COMMAND git status --porcelain --untracked-files=no
    WORKING_DIRECTORY "${OMEGA_LLAMA_SOURCE}" OUTPUT_VARIABLE LLAMA_DIRTY
    OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE LLAMA_STATUS_RESULT ERROR_QUIET)
if(NOT LLAMA_STATUS_RESULT EQUAL 0 OR NOT "${LLAMA_DIRTY}" STREQUAL "")
    message(FATAL_ERROR "Refusing modified llama.cpp source tree")
endif()
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
add_subdirectory("${OMEGA_LLAMA_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/llama" EXCLUDE_FROM_ALL)
add_library(omega_llama_jni SHARED OmegaLlamaEngine.cpp OmegaLlamaJni.cpp)
target_compile_features(omega_llama_jni PRIVATE cxx_std_20)
target_include_directories(omega_llama_jni PRIVATE "${OMEGA_LLAMA_SOURCE}/include" "${OMEGA_LLAMA_SOURCE}/ggml/include")
target_link_libraries(omega_llama_jni PRIVATE llama)
target_compile_options(omega_llama_jni PRIVATE -Wall -Wextra -Werror)
target_link_options(omega_llama_jni PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
